func phase4StartGatewayPortForward(t *testing.T, localPort string) func() {
	t.Helper()
	cmd := exec.Command("kubectl", "port-forward",
		"-n", "ach-system",
		"svc/ach-local-gateway",
		localPort+":80",
	)
	if err := cmd.Start(); err != nil {
		t.Skipf("phase4StartGatewayPortForward: cannot start port-forward: %v", err)
		return func() {}
	}
	// cmd.Start() only fails on fork/exec errors — it returns nil even when
	// kubectl exits milliseconds later (port in use, RBAC denial, missing
	// Service). Probe the local end with a bounded TCP dial so a dead
	// port-forward skips cleanly instead of failing a downstream HTTP call
	// with a confusing "connection refused". Mirrors startPortForward in
	// phase2_sc5_orphan_test.go.
	deadline := time.Now().Add(30 * time.Second)
	for time.Now().Before(deadline) {
		conn, err := net.DialTimeout("tcp", "127.0.0.1:"+localPort, 500*time.Millisecond)
		if err == nil {
			_ = conn.Close()
			return func() { _ = cmd.Process.Kill() }
		}
		time.Sleep(200 * time.Millisecond)
	}
	_ = cmd.Process.Kill()
	t.Skipf("phase4StartGatewayPortForward: port-forward never listened on 127.0.0.1:%s within 30s", localPort)
	return func() {}
}
```

## Task 4: #36 — return authoritative `u.UserID`, drop unguarded placeholder swap

**Why:** In `UserInfoByEmail`'s `/user/list` fallback, the second `/user/info?user_id=` call returns `idEnv.UserID` with NO `default_user_id` guard (the email path has one). If LiteLLM returns the placeholder (or empty) on the by-id path, the authoritative `u.UserID` from `/user/list` is silently overwritten — downstream `POST /team/member_add` and `POST /key/generate` then mint keys against no user or the shared admin pseudo-user. The by-id call's only useful contribution is the `TeamAlias` mapping; keep that, use `u.UserID`. Also remove the dead `UserInfo struct { UserEmail string }` field.

			// (user_api_key_auth_mcp.py) nullifies any x-litellm-api-key
			// lacking a "Bearer " prefix. /v1, /gemini, and /a2a validate
			// the bare master key, so the prefix is scoped to /mcp here
			// rather than baked into the shared StripAndRewrite (#41).
			if deps.LiteLLMMasterKey != "" && routeFor(req.URL.Path) == "/mcp" {
				req.Header.Set("X-Litellm-Api-Key", "Bearer "+deps.LiteLLMMasterKey)
			}
```

```go
// ExternalRefRefreshDeps bundles the per-reconciler dependencies the
// shared materializeExternalRef needs. Plugin/Prompt/Artifact reconcilers
// build one per Reconcile call.
type ExternalRefRefreshDeps struct {
    Client       client.Client          // for Secret Get
    Namespace    string                 // for Secret resolution
    DB           *pgxpool.Pool          // for UpsertExternalRef
    CacheRoot    string                 // for .tmp/ + final path roots
    Kind         string                 // "plugin" | "prompt" | "artifact" — drives external_refs.kind column
    Name         string                 // CR metadata.name — drives external_refs.name column AND final path leaf
    SourceSpec   sources.SourceSpec     // built from cr.Spec by the per-kind reconciler
    AuthSecretRef *achv1alpha1.SourceAuthSecretRef  // nil-safe (HTTPSource.AuthSecretRef is *)
    Refresh      achv1alpha1.RefreshBlock
    PriorRev     string                 // fetched from external_refs.upstream_rev for conditional GET
    SizeCapBytes int64                  // 0 = no cap (Prompt/Artifact); >0 = enforce LimitReader (Plugin)
    FinalPath    string                 // computed by the per-kind reconciler before calling
    Fetchers     FetcherFactory         // nil → use sources.For
    Log          logr.Logger
}

// MaterializeResult is what materializeExternalRef returns to the per-kind reconciler.
type MaterializeResult struct {
    UpstreamRev string                  // empty when NotModified=true (caller doesn't UPSERT)
    NotModified bool                    // true when conditional-GET returned 304
    Err         error                   // nil on success/NotModified; non-nil with sources.ReasonOf-classifiable err on failure
}

// materializeExternalRef runs §10.3 steps 1-6 for one CR:
//   1. Resolve auth Secret via client.Get.
//   2. sources.For(spec) → Fetcher.
//   3. fetcher.Fetch(ctx, FetchRequest{Spec, Secret, PriorRev: deps.PriorRev}).
//   4. If NotModified → return {NotModified: true, UpstreamRev: deps.PriorRev}.
//   5. Stage in deps.CacheRoot/.tmp/<random>.
//   6. Wrap body in io.LimitReader if SizeCapBytes > 0; io.Copy to staging; check overshoot; fsync; rename(2) to deps.FinalPath.
//   7. db.UpsertExternalRef.
//   8. Return {UpstreamRev: result.UpstreamRev}.
//
// Returns an err shaped for sources.ReasonOf classification (or the
// special OversizeError for Plugin). Caller maps reason to status condition.
func materializeExternalRef(ctx context.Context, deps ExternalRefRefreshDeps) MaterializeResult
```

```
// TargetIndexKey is the controller-runtime field-indexer key used to look up
// BackendIdentityPolicy CRs by their target {kind, name} tuple. Format of
// the indexed string: "<spec.target.kind>/<spec.target.name>"
// (e.g. "MCPServer/github-mcp", "A2AAgent/openai-a2a").
const TargetIndexKey = "spec.target"

// RegisterIndex teaches the controller-runtime cache to index
// BackendIdentityPolicy by "<kind>/<name>" so request-time lookups via
// List(... MatchingFields{TargetIndexKey: "MCPServer/foo"}) hit an
// O(log N) index instead of a full namespace scan.
//
// CONTRACT: This MUST be called AFTER ctrl.NewManager and BEFORE the
// first GetInformer call on BackendIdentityPolicy — controller-runtime
// rejects late-registered indexers.
func RegisterIndex(ctx context.Context, mgr ctrl.Manager) error {
    return mgr.GetFieldIndexer().IndexField(ctx, &achv1alpha1.BackendIdentityPolicy{},
        TargetIndexKey,
        func(obj client.Object) []string {
            b := obj.(*achv1alpha1.BackendIdentityPolicy)
            return []string{string(b.Spec.Target.Kind) + "/" + b.Spec.Target.Name}
        })
}

// ResolveWinner returns the alphabetically-LAST BackendIdentityPolicy for
// the given (kind, name) target. Returns nil when:
//   - zero BIPs match, OR
//   - the alpha-LAST winner has spec.forwardIdentityJWT=false
//     (explicit opt-out is equivalent to no policy)
//
// Per TODO.md §6 + FWD-05 + Hub §9.3: multiple BIPs targeting the same
// (kind, name) tuple coexist. Operators flip precedence by renaming CRs
// (a "zz-" prefix on metadata.name makes the rename the alpha-LAST winner).
// There is NO DuplicateTarget status reason; the Operator stays dumb.
//
// Per OP-16: this function reads spec.target + spec.forwardIdentityJWT
// ONLY — it MUST NOT read .status (Operator is the sole status writer
// and runtime authority is decoupled from status-write latency).
func ResolveWinner(ctx context.Context, c client.Client, kind, name, namespace string) *achv1alpha1.BackendIdentityPolicy {
    var list achv1alpha1.BackendIdentityPolicyList
    if err := c.List(ctx, &list,
        client.MatchingFields{TargetIndexKey: kind + "/" + name},
        client.InNamespace(namespace)); err != nil {
        return nil
    }
    if len(list.Items) == 0 { return nil }
    sort.SliceStable(list.Items, func(i, j int) bool {
        return list.Items[i].Name < list.Items[j].Name
    })
    winner := list.Items[len(list.Items)-1]
    if !winner.Spec.ForwardIdentityJWT { return nil }
    return &winner
}
```

NEVER read `winner.Status` in this function (and never refer to `Synced=DuplicateTarget` anywhere in source comments — Phase 4 surface is `DuplicateTarget`-free).

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| BIP CR author → Forwarder authorization decision | An operator who can write BIPs can choose whether downstream backends receive an ACH JWT. RBAC at the K8s layer governs who can create BIPs. |
| Forwarder informer → cluster API server | controller-runtime cache; namespace-scoped; no cross-namespace authority. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-04-01 | Elevation of privilege | BIP rename attack | accept | Anyone with `update` on backendidentitypolicies in the namespace can rename CRs to flip the alpha-LAST winner. Documented per TODO §6 — this IS the intended administration mechanism. Mitigated at K8s RBAC layer (only platform admins should have write on BIPs in the deployment namespace). |
| T-04-04-02 | Spoofing | Stale informer cache | mitigate | controller-runtime cache eventually-consistent; bounded by the namespace-watch latency (sub-second in practice). Hub spec accepts this window; no synchronous Get path. Plan 04-08 readiness probe gates on WaitForCacheSync so initial sync completes before any request is served. |
| T-04-04-03 | Tampering | Direct Status read | mitigate | Source contract: ResolveWinner reads spec only, NEVER .Status (per OP-16). Test B12 enforces via grep on source code. |
| T-04-04-04 | Information disclosure | Index key leak | accept | TargetIndexKey value `"spec.target"` is non-secret. The indexed string `<kind>/<name>` is also non-secret (BIPs themselves are visible to anyone with list permission on the namespace). |
| T-04-04-05 | Denial of service | Pathological BIP count | accept | A namespace with N BIPs targeting the same (kind, name) costs O(N log N) per request due to sort. Hub spec is silent on per-namespace BIP cap; deployment-layer concern. Adding a cap would change runtime behavior in ways the spec doesn't anticipate. |
| T-04-04-SC | Tampering | go module install | mitigate | No new go.mod entries; uses existing controller-runtime + apimachinery + project-internal v1alpha1 types. |

</threat_model>

func testPhase4SC1HeaderRewrite(t *testing.T) {
    // Acquire pk_ via Phase 3 SSO flow (reuse phase3_helpers acquirePk if available).
    pk := mustAcquirePk(t)
    // POST to /v1/chat/completions with the pk_ + a fake Authorization header.
    // Expected: 200 from the LiteLLM upstream (or the test mock if LiteLLM
    // refuses without a real model). Verify via the mock-side capture (when
    // available) that the forwarded request has:
    //   x-litellm-api-key == <shared key>
    //   x-litellm-key-id  == <litellm_token from KeyInfo>
    //   NO Authorization
    //   NO x-ach-*
    //
    // If a LiteLLM-side capture API is not available in the kind cluster,
    // assert at least the response status is NOT 401 from LiteLLM (which
    // would indicate ACH failed to write the shared-key headers).
    forwarderURL := os.Getenv("ACH_FORWARDER_URL")
    if forwarderURL == "" { t.Skip("ACH_FORWARDER_URL not set") }
    req, _ := http.NewRequest("POST", forwarderURL+"/v1/chat/completions",
        strings.NewReader(`{"model":"test-model","messages":[{"role":"user","content":"hi"}]}`))
    req.Header.Set("x-ach-key", pk)
    req.Header.Set("authorization", "Bearer should-be-stripped")
    req.Header.Set("x-litellm-api-key", "should-be-stripped-and-overwritten")
    req.Header.Set("x-ach-environment", "should-be-stripped")
    resp, err := http.DefaultClient.Do(req)
    if err != nil { t.Fatalf("forwarder request: %v", err) }
    defer resp.Body.Close()
    body, _ := io.ReadAll(resp.Body)
    // The Phase 4 forwarder must NOT echo the request body verbatim; what
    // we get back is whatever LiteLLM returned. A 401 from LiteLLM would
    // suggest the shared key wasn't attached.
    if resp.StatusCode == http.StatusUnauthorized {
        t.Fatalf("LiteLLM returned 401 — header rewrite likely failed. Body: %s", string(body))
    }
    // Pass on any 2xx or 4xx other than 401 (LiteLLM may legitimately reject
    // an unknown model with 404 or similar; that's fine — header rewrite
    // succeeded if we got past LiteLLM's auth).
    t.Logf("SC1 forwarder→LiteLLM round-trip status: %d (header rewrite verified at LiteLLM auth-pass)", resp.StatusCode)
}

func testPhase4SC2McpA2aPrecheck(t *testing.T) {
    ek := mustAcquireEkBoundToEnv(t, "demo")  // bound to env "demo" with mcpServers=["allowed"]
    forwarderURL := os.Getenv("ACH_FORWARDER_URL")

    // ek_ to /mcp/disallowed → 403 unauthorized_resource (name not in bound env).
    req, _ := http.NewRequest("GET", forwarderURL+"/mcp/disallowed/", nil)
    req.Header.Set("x-ach-key", ek)
    resp, err := http.DefaultClient.Do(req)
    if err != nil { t.Fatalf("disallowed mcp request: %v", err) }
    body, _ := io.ReadAll(resp.Body)
    resp.Body.Close()
    if resp.StatusCode != http.StatusForbidden {
        t.Fatalf("expected 403 unauthorized_resource on /mcp/disallowed; got %d body=%s", resp.StatusCode, string(body))
    }
    var env struct {
        Error struct{ Code string `json:"code"` } `json:"error"`
    }
    if err := json.Unmarshal(body, &env); err != nil { t.Fatalf("envelope decode: %v", err) }
    if env.Error.Code != "unauthorized_resource" {
        t.Fatalf("expected code=unauthorized_resource; got %q", env.Error.Code)
    }

    // ek_ to /mcp/allowed → some non-403 status (LiteLLM 404 acceptable;
    // means precheck passed and request was forwarded).
    reqAllowed, _ := http.NewRequest("GET", forwarderURL+"/mcp/allowed/", nil)
    reqAllowed.Header.Set("x-ach-key", ek)
    respAllowed, err := http.DefaultClient.Do(reqAllowed)
    if err != nil { t.Fatalf("allowed mcp request: %v", err) }
    respAllowed.Body.Close()
    if respAllowed.StatusCode == http.StatusForbidden {
        t.Fatalf("expected non-403 on /mcp/allowed; precheck rejected an allowed name")
    }

    // pk_ to /mcp/allowed with NO team intersection → 403 unauthorized_team.
    // (Use a pk_ owned by a user that's NOT in env.authorizedTeams.)
    pkUnaffiliated := mustAcquirePkUnaffiliated(t)
    reqPk, _ := http.NewRequest("GET", forwarderURL+"/mcp/allowed/", nil)
    reqPk.Header.Set("x-ach-key", pkUnaffiliated)
    respPk, _ := http.DefaultClient.Do(reqPk)
    bodyPk, _ := io.ReadAll(respPk.Body)
    respPk.Body.Close()
    if respPk.StatusCode != http.StatusForbidden {
        t.Fatalf("expected 403 unauthorized_team for unaffiliated pk_; got %d body=%s", respPk.StatusCode, string(bodyPk))
    }
}

// testPhase4SC2EkTagInjection verifies FWD-06 v1alpha1 scope:
//
//   * ek_ POST to /v1/chat/completions → upstream request body contains
//     metadata.tags with "environment:<envName>" appended.
//   * pk_ POST to the same endpoint → upstream request body unchanged
//     (no environment: tag added).
//
// Requires ACH_MCP_MOCK_CAPTURE_URL OR a dedicated tag-capture endpoint
// on the kind cluster's LiteLLM mock. If the mock supports raw-body
// capture via a /capture endpoint that returns the last received body
// as JSON, we read it back; otherwise t.Skipf with engineer-pending.
//
// Mock contract (matches the Plan 04-08 helm template's optional
// `litellmMock` Service when enabled): POST /v1/chat/completions echoes
// 200 with body `{"id":"mock"}`; GET /capture returns the LAST captured
// request body as a JSON string field `{"last_body": "<raw>"}`.
func testPhase4SC2EkTagInjection(t *testing.T) {
    captureURL := os.Getenv("ACH_LITELLM_MOCK_CAPTURE_URL")
    if captureURL == "" {
        t.Skipf("ACH_LITELLM_MOCK_CAPTURE_URL not set — cannot verify FWD-06 upstream body capture. Run with mock LiteLLM that exposes /capture. Skipping (engineer-pending).")
    }
    forwarderURL := os.Getenv("ACH_FORWARDER_URL")
    if forwarderURL == "" {
        t.Skip("ACH_FORWARDER_URL not set")
    }

    body := `{"model":"test-model","messages":[{"role":"user","content":"hi"}]}`

    // ek_ branch — expect tag injection.
    ek := mustAcquireEkBoundToEnv(t, "demo")
    reqEk, _ := http.NewRequest("POST", forwarderURL+"/v1/chat/completions", strings.NewReader(body))
    reqEk.Header.Set("x-ach-key", ek)
    reqEk.Header.Set("Content-Type", "application/json")
    respEk, err := http.DefaultClient.Do(reqEk)
    if err != nil {
        t.Fatalf("ek_ /v1 request: %v", err)
    }
    respEk.Body.Close()

    // Read upstream-captured body from the mock.
    captured := mustGetCapturedBody(t, captureURL)
    var doc map[string]any
    if err := json.Unmarshal([]byte(captured), &doc); err != nil {
        t.Fatalf("captured ek_ body not JSON: %v\nbody=%s", err, captured)
    }
    meta, ok := doc["metadata"].(map[string]any)
    if !ok {
        t.Fatalf("FWD-06 ek_: metadata missing from upstream body\nbody=%s", captured)
    }
    tags, ok := meta["tags"].([]any)
    if !ok {
        t.Fatalf("FWD-06 ek_: metadata.tags missing or not array\nbody=%s", captured)
    }
    found := false
    for _, tg := range tags {
        if s, ok := tg.(string); ok && strings.HasPrefix(s, "environment:") {
            found = true
            break
        }
    }
    if !found {
        t.Fatalf("FWD-06 ek_: expected metadata.tags to contain 'environment:<name>'; got tags=%v", tags)
    }

    // pk_ branch — expect NO tag injection.
    pk := mustAcquirePk(t)
    reqPk, _ := http.NewRequest("POST", forwarderURL+"/v1/chat/completions", strings.NewReader(body))
    reqPk.Header.Set("x-ach-key", pk)
    reqPk.Header.Set("Content-Type", "application/json")
    respPk, err := http.DefaultClient.Do(reqPk)
    if err != nil {
        t.Fatalf("pk_ /v1 request: %v", err)
    }
    respPk.Body.Close()

    capturedPk := mustGetCapturedBody(t, captureURL)
    var docPk map[string]any
    if err := json.Unmarshal([]byte(capturedPk), &docPk); err != nil {
        t.Fatalf("captured pk_ body not JSON: %v\nbody=%s", err, capturedPk)
    }
    if metaPk, ok := docPk["metadata"].(map[string]any); ok {
        if tagsPk, ok := metaPk["tags"].([]any); ok {
            for _, tg := range tagsPk {
                if s, ok := tg.(string); ok && strings.HasPrefix(s, "environment:") {
                    t.Fatalf("FWD-06 pk_ exemption violated: found 'environment:' tag in pk_ upstream body; tags=%v", tagsPk)
                }
            }
        }
    }
}

func testPhase4SC4JwksAndSecretRbac(t *testing.T) {
    baseURL := os.Getenv("ACH_BASE_URL")
    resp, err := http.Get(baseURL + "/.well-known/jwks.json")
    if err != nil { t.Fatalf("jwks GET: %v", err) }
    defer resp.Body.Close()
    if resp.StatusCode != 200 { t.Fatalf("jwks status %d", resp.StatusCode) }
    if ct := resp.Header.Get("Content-Type"); ct != "application/jwk-set+json" {
        t.Fatalf("Content-Type want=application/jwk-set+json got=%q", ct)
    }
    if cc := resp.Header.Get("Cache-Control"); cc != "public, max-age=3600" {
        t.Fatalf("Cache-Control want='public, max-age=3600' got=%q", cc)
    }
    body, _ := io.ReadAll(resp.Body)
    var doc struct {
        Keys []struct {
            Kty, Crv, Use, Alg, Kid, X string
        } `json:"keys"`
    }
    if err := json.Unmarshal(body, &doc); err != nil { t.Fatalf("decode jwks: %v", err) }
    if len(doc.Keys) == 0 { t.Fatal("jwks empty") }
    for i, k := range doc.Keys {
        if k.Kty != "OKP" { t.Errorf("keys[%d].kty want=OKP got=%q", i, k.Kty) }
        if k.Crv != "Ed25519" { t.Errorf("keys[%d].crv want=Ed25519 got=%q", i, k.Crv) }
        if k.Kid == "" { t.Errorf("keys[%d].kid empty", i) }
        if k.X == "" { t.Errorf("keys[%d].x empty", i) }
    }

    // Negative RBAC test: a non-forwarder ServiceAccount cannot read the JWT Secret.
    // Use the operator ServiceAccount (which has no resourceNames carve-out for ach-jwt-signing-keys).
    nsOut := mustKubectlGetOutput(t, "secret", "ach-jwt-signing-keys", "--as=system:serviceaccount:ach-system:ach-operator", "-n", "ach-system")
    if !strings.Contains(nsOut, "forbidden") && !strings.Contains(nsOut, "not found") && !strings.Contains(nsOut, "cannot") {
        // The exact phrasing varies by kubectl version. If kubectl exits 0 with the Secret body,
        // RBAC is over-permissioned — fail.
        t.Fatalf("ach-operator ServiceAccount should NOT be able to read ach-jwt-signing-keys; got: %s", nsOut)
    }
}

func testPhase4SC5RefuseToStartOnNonHttpsBaseURL(t *testing.T) {
    // This test is intentionally skip-with-engineer-pending unless the cluster
    // pre-arranged a dedicated namespace with a broken BASE_URL Deployment.
    // The runtime gate is the FWD-10 refuse-to-start in cmd/ach/cmd/forwarder.go
    // (validateForwarderConfig); see Plan 04-08 acceptance criteria for the unit-test
    // coverage. This e2e subtest verifies the same gate in a live K8s context.
    if os.Getenv("ACH_E2E_PHASE4_SC5_NS") == "" {
        t.Skip("SC5 requires a dedicated namespace with helm install --set forwarder.baseUrl=http://invalid; set ACH_E2E_PHASE4_SC5_NS=<ns> to run. Skipping (engineer-pending).")
    }
    ns := os.Getenv("ACH_E2E_PHASE4_SC5_NS")

    // CLAUDE.md "Bash anti-patterns" — naked polling banned.
    // Capture pod name ONCE; assert non-empty so the loop never starts
    // against a vanished target. Use --previous on kubectl logs so we
    // read the crashed container's logs (CrashLoopBackOff terminates the
    // container; live logs would return empty / error).
    podOut, err := exec.Command("kubectl", "-n", ns, "get", "pods",
        "-l", "app.kubernetes.io/component=forwarder",
        "-o", "jsonpath={.items[0].metadata.name}").Output()
    if err != nil {
        t.Fatalf("SC5: kubectl get pods failed: %v", err)
    }
    podName := strings.TrimSpace(string(podOut))
    if podName == "" {
        t.Fatalf("SC5: no forwarder pod found in namespace %s", ns)
    }

    // Bounded retry with explicit deadline + explicit failure exit
    // (CLAUDE.md form 3). Per-iteration sleep stays small (5s), max
    // wall-clock 60s. Read --previous logs since the container has
    // already crashed; current container logs would be empty/error.
    deadline := time.Now().Add(60 * time.Second)
    var lastLogs string
    for time.Now().Before(deadline) {
        logsOut, logsErr := exec.Command("kubectl", "-n", ns, "logs",
            podName, "--previous", "--tail=50").CombinedOutput()
        if logsErr == nil {
            lastLogs = string(logsOut)
            if strings.Contains(lastLogs, "ACH_BASE_URL must be https://") {
                return // PASS — refuse-to-start message observed.
            }
        }
        time.Sleep(5 * time.Second)
    }
    t.Fatalf("SC5: FWD-10 refuse-to-start message not seen within 60s for pod %s/%s\nLast logs:\n%s",
        ns, podName, lastLogs)
}

Create `test/e2e/phase4_helpers_test.go` (build tag `//go:build e2e`) with the helpers:
- `waitForwarderReady(t, timeout) error` — calls `kubectl -n ach-system rollout status deploy/ach-forwarder --timeout=<timeout>`; returns error on non-zero exit.
- `mustAcquirePk(t)` / `mustAcquireEkBoundToEnv(t, envName)` / `mustAcquirePkUnaffiliated(t)` — reuse Phase 3 helpers if signature matches; else thin wrappers.
- `mustApplyFixture(t, path)` / `mustDeleteFixture(t, path)` — kubectl apply -f / kubectl delete -f.
- `mustKubectlPatch(t, target, jsonPatch)` — kubectl patch with `--type=merge`.

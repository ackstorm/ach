**Net-new pattern to write** (no analog; this is the contract):
```go
package main

import (
    "log/slog"
    "net/http"
    "os"
    "os/signal"
    "syscall"
)

func main() {
    logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))
    logger.Info("Phase 1 stub: <binary> starting", "phase", 1)

    mux := http.NewServeMux()
    mux.HandleFunc("/healthz", func(w http.ResponseWriter, _ *http.Request) {
        w.WriteHeader(http.StatusOK)
    })
    srv := &http.Server{Addr: ":8081", Handler: mux}

    go func() {
        if err := srv.ListenAndServe(); err != nil && err != http.ErrServerClosed {
            logger.Error("server error", "err", err)
            os.Exit(1)
        }
    }()

    sig := make(chan os.Signal, 1)
    signal.Notify(sig, syscall.SIGINT, syscall.SIGTERM)
    <-sig
    _ = srv.Shutdown(nil)
}
```

**Critical for Phase 1 SC #2:** Content Service stub MUST be a long-running healthy process — a sidecar that immediately exits fails the "two containers ready" check.

### `internal/credhash/credhash.go` (utility, transform)

**Analog:** NONE — `analog: NONE; rationale: HMAC-SHA-256 + constant-time compare is a stdlib pattern (crypto/hmac + crypto/subtle); sister has no credential hashing. Hub §16.1 specifies HMAC-SHA-256 with server-side pepper; D-09 specifies env var ACH_CREDENTIAL_HASH_PEPPER as the pepper source.`

**Net-new pattern to write** (stdlib only):
```go
package credhash

import (
    "crypto/hmac"
    "crypto/sha256"
    "crypto/subtle"
    "encoding/hex"
    "errors"
)

var ErrEmptyPepper = errors.New("credhash: pepper is empty")

// Hash computes HMAC-SHA-256(pepper, plaintext) returning hex-encoded digest.
// Plaintext is the full bearer key (pk_... or ek_...).
func Hash(pepper, plaintext []byte) (string, error) {
    if len(pepper) == 0 {
        return "", ErrEmptyPepper
    }
    h := hmac.New(sha256.New, pepper)
    h.Write(plaintext)
    return hex.EncodeToString(h.Sum(nil)), nil
}

// Equal performs constant-time comparison of two hex digests.
// Returns false on any decode failure (never panics).
func Equal(a, b string) bool {
    ab, errA := hex.DecodeString(a)
    bb, errB := hex.DecodeString(b)
    if errA != nil || errB != nil {
        return false
    }
    return subtle.ConstantTimeCompare(ab, bb) == 1
}
```

**Test pattern (stdlib `testing` only):** Validates the contract from DB-03 — same input + same pepper → same hash; different peppers → different hashes; empty pepper → error; constant-time compare returns true/false correctly.

// EnsureLayout creates the OP-10 cache directory tree under root.

package config

import (
    "errors"
    "os"
    "strconv"
)

func EnvOr(key, fallback string) string {
    if v := os.Getenv(key); v != "" {
        return v
    }
    return fallback
}

func EnvBool(key string, fallback bool) bool {
    v := os.Getenv(key)
    if v == "" {
        return fallback
    }
    b, err := strconv.ParseBool(v)
    if err != nil {
        return fallback
    }
    return b
}

// MustEnvNonEmpty returns os.Getenv(key) or returns an error when empty.
// Used for ACH_DB_URL and ACH_CREDENTIAL_HASH_PEPPER (D-08, D-09).
func MustEnvNonEmpty(key string) (string, error) {
    v := os.Getenv(key)
    if v == "" {
        return "", errors.New("config: " + key + " is empty or unset")
    }
    return v, nil
}

// MustEnvInt parses an int from key. Returns error on 0/negative/non-numeric.
// Used for ACH_PLUGIN_MAX_SIZE_MIB (OP-09; refuses to start at 0/negative).
func MustEnvIntPositive(key string, fallback int) (int, error) {
    v := os.Getenv(key)
    if v == "" {
        return fallback, nil
    }
    n, err := strconv.Atoi(v)
    if err != nil || n <= 0 {
        return 0, errors.New("config: " + key + " must be a positive integer")
    }
    return n, nil
}

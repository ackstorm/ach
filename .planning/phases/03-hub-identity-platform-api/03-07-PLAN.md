    ```
        // Generate state: 16 random bytes -> base64url.
        stateBytes := make([]byte, 16)

        http.Redirect(w, r, authURL, http.StatusFound)
    }
    ```

    ```
        state, verifier, err := readSSOCookie(r)
        if err != nil { audit + render.Error 400; return }
        clearSSOCookie(w)  // always clear after read

        urlState := r.URL.Query().Get("state")
        // BLK-05 commit: missing OR mismatching URL state -> 400 invalid_argument with OutcomeStateInvalid.
        // r.URL.Query().Get returns "" for both "absent" and "?state=" (empty-value) — both cases hit this branch.
        if urlState == "" || urlState != state {
            audit.EmitAudit(ctx, deps.Audit, audit.Event{
                Action: audit.ActionSSOLogin,
                Outcome: audit.OutcomeStateInvalid,
                Actor: actor, RequestID: reqID,
            })
            render.Error(w, 400, audit.OutcomeStateInvalid, "state_invalid", reqID)
            return
        }

        code := r.URL.Query().Get("code")
        if code == "" { render.Error 400; return }

        // 2. Exchange code for token.
        token, err := deps.OAuth2Cfg.Exchange(ctx, code, oauth2.SetAuthURLParam("code_verifier", verifier))
        if err != nil { audit OutcomeInternalError + render.Error 502; return }

        // 3. Extract + verify ID token.
        rawIDToken, ok := token.Extra("id_token").(string)
        if !ok { audit + render.Error 502; return }

        verifier_ := deps.OIDCProvider.Verifier(&oidc.Config{ClientID: deps.OAuth2Cfg.ClientID})
        idToken, err := verifier_.Verify(ctx, rawIDToken)
        if err != nil { audit OutcomeInternalError + render.Error 401; return }

        var claims struct {
            Email string `json:"email"`
        }
        if err := idToken.Claims(&claims); err != nil { audit + render.Error 500; return }
        if claims.Email == "" { audit + render.Error 400; return }

        actor := deps.Namespace + "/" + claims.Email

        // 4. Idempotent LiteLLM user provision.
        user, err := deps.LiteLLM.UserInfoByEmail(ctx, claims.Email)
        var userID string
        switch {
        case err != nil && errors.Is(err, litellm.ErrNotFound):
            // First-time SSO: create the user.
            user, err = deps.LiteLLM.UserNew(ctx, &litellm.UserNewRequest{UserEmail: claims.Email, Teams: []string{"default"}})
            if err != nil { audit OutcomeLitellmUnreachable + render.Error 503; return }
            userID = user.UserID
            // TeamMemberAdd: per D-04 step 5, MUST succeed; failure on "default" means deployer misconfig.
            if err := deps.LiteLLM.TeamMemberAdd(ctx, "default", userID, "user"); err != nil {
                // Classify error: 4xx with team_not_found vs transport vs other.
                // For Phase 3, treat any TeamMemberAdd error after a successful UserNew as default_team_missing.
                emitAudit(audit.OutcomeDefaultTeamMissing, actor, reqID)
                render.Error(w, 500, "default_team_missing", "...", reqID)
                return
            }
        case err != nil:
            // Transport / non-404 error.
            audit OutcomeLitellmUnreachable + render.Error 503; return
        default:
            // Existing user. BLK-05 sub-point 3 + D-25: always call TeamMemberAdd to be idempotent;
            // LiteLLM treats duplicate-add as a 4xx that the caller swallows. This makes the SSO flow
            // safe against a future state where the user's "default" Team membership was revoked
            // out-of-band (e.g., admin manually removed). The 4xx duplicate-add is logged but not
            // treated as default_team_missing.
            userID = user.UserID
            if err := deps.LiteLLM.TeamMemberAdd(ctx, "default", userID, "user"); err != nil {
                // Classify: if the error indicates "default Team missing" (4xx with team_not_found body
                // OR any transport/5xx that LiteLLM SHOULD have returned for a missing team), fail loud.
                // Duplicate-add 4xx (D-25): swallowed silently (the membership already exists — desired
                // state). Phase 3 cannot reliably distinguish duplicate-add from team-not-found without
                // parsing the LiteLLM error body, so the heuristic is:
                //   - litellm.ErrNotFound (404 from team lookup) -> default_team_missing.
                //   - other non-transient error -> default_team_missing (fail-loud per Hub §17 / API-02).
                //   - transient error (network/5xx) -> 503 litellm_unreachable.
                // The executor reads the SUMMARY of Plan 03-07 from a prior run (if any) to refine the
                // classifier; for Phase 3 ship, the safest default is fail-loud on any error from the
                // existing-user TeamMemberAdd to preserve the API-02 invariant.
                deps.Logger.Warn("sso.callback: TeamMemberAdd on existing-user path failed", "err", err)
                emitAudit(audit.OutcomeDefaultTeamMissing, actor, reqID)
                render.Error(w, 500, "default_team_missing", "default Team unreachable", reqID)
                return
            }
        }

        // 5. Mint pk_ + pkid_ + hash + LiteLLM key.
        plaintext, err := keys.NewBearer(keys.PrefixPk)
        if err != nil { audit + render.Error 500; return }
        keyID, err := keys.NewKeyID(keys.PrefixPkid)
        if err != nil { audit + render.Error 500; return }

        credentialHash, err := credhash.Hash(deps.Pepper, []byte(plaintext))
        if err != nil { audit + render.Error 500; return }

        keyResp, err := deps.LiteLLM.KeyGenerate(ctx, &litellm.KeyGenerateRequest{
            UserID:    userID,
            Key:       plaintext,
            MaxBudget: nil,  // KEY-10 invariant
        })
        if err != nil { audit OutcomeLitellmUnreachable + render.Error 503; return }

        // 6. INSERT row.
        expiresAt := time.Now().Add(7 * 24 * time.Hour)
        row := db.PkInsertRow{
            KeyID:          keyID,
            CredentialHash: credentialHash,
            OwnerEmail:     claims.Email,
            Status:         "active",
            ExpiresAt:      expiresAt,
            LiteLLMUserID:  &userID,
            LiteLLMToken:   &keyResp.Token,
        }
        if err := db.InsertPersonalKey(ctx, deps.Pool, row); err != nil {
            // Compensation: revoke the LiteLLM-side key we just created.
            if cleanupErr := deps.LiteLLM.RevokeKey(context.Background(), keyResp.Token); cleanupErr != nil {
                deps.Logger.Error("sso.callback: compensation revoke failed", "err", cleanupErr)
            }
            audit OutcomeDbInsertFailed + render.Error 500; return
        }

        // 7. Emit audit success.
        audit.EmitAudit(ctx, deps.Audit, audit.Event{
            Action:    audit.ActionSSOLogin,
            Outcome:   audit.OutcomeCreated,
            Actor:     actor,
            RequestID: reqID,
            KeyID:     keyID,
        })

        // 8. Return JSON {key_id, plaintext, owner_email}.
        render.JSON(w, 200, map[string]string{
            "key_id":      keyID,
            "plaintext":   plaintext,
            "owner_email": claims.Email,
        })

        // After render.JSON, plaintext goes out of scope (no further reference in this handler).
    }
    ```
